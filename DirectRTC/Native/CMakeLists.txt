cmake_minimum_required(VERSION 3.20)
project(DotNetEq LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${DOT_ABSEIL_SOURCE}" abseil)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
add_subdirectory("${DOT_OPUS_SOURCE}" opus)
set(W "${DOT_WEBRTC_SOURCE}")
include(neteq_sources.cmake)
add_library(neteq_upstream STATIC ${WEBRTC_SOURCES})
target_include_directories(neteq_upstream PUBLIC "${W}" "${DOT_GENERATED_INCLUDE}" "${DOT_ABSEIL_SOURCE}")
target_compile_definitions(neteq_upstream PUBLIC WEBRTC_MAC WEBRTC_POSIX WEBRTC_CODEC_OPUS WEBRTC_STRICT_FIELD_TRIALS=0 WEBRTC_OPUS_SUPPORT_120MS_PTIME=1 WEBRTC_ENABLE_SYMBOL_EXPORT=0 WEBRTC_NON_STATIC_TRACE_EVENT_HANDLERS=0)
target_compile_options(neteq_upstream PRIVATE -Wno-deprecated-declarations -Wno-nullability-completeness)
target_link_libraries(neteq_upstream PUBLIC opus absl::strings absl::string_view absl::flat_hash_map absl::optional absl::any_invocable absl::synchronization)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS" OR CMAKE_SYSTEM_NAME STREQUAL "watchOS")
  target_compile_definitions(neteq_upstream PUBLIC WEBRTC_IOS)
endif()
add_library(DotNetEq STATIC DotNetEqBridge.cc)
target_include_directories(DotNetEq PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(DotNetEq PRIVATE neteq_upstream)
